# simulate-mutators.txt — the commands TAPPAAS_SIMULATE=1 does not run. The one
# list: lib/ts/src/exec.ts (mutators) and common-install-routines.sh both read it.
#
# Format: <command> <verb>...   # why
# A verb is the command's first word that is not an option. `*` is any verb.
# `verb:sub` is the first two such words; a word starting with `-` is an option
# found anywhere in the command, for tools without verbs. `sudo` lists what is
# held behind it on top of what is held anyway.
# What is not listed runs, so `pvesh get`, `qm status`, `zfs list`, `zpool list`,
# `systemctl is-active` give a preview its real answers. A command found behind
# sudo, env, timeout, `bash -c` or ssh counts the same.
pvesh set create delete                                        # Proxmox API writes; `get` and `ls` read
qm start stop shutdown reboot reset suspend resume             # guest power state
qm destroy create set clone template                           # guest existence and config
qm migrate remote-migrate                                      # moves a guest
qm snapshot rollback delsnapshot                               # guest snapshots
qm importdisk disk resize move-disk move_disk unlink           # guest disks
pct start stop shutdown reboot suspend resume                  # container power state
pct destroy create set clone template restore                  # container existence and config
pct migrate remote-migrate                                     # moves a container
pct snapshot rollback delsnapshot                              # container snapshots
pct resize move-volume mount unmount push                      # container volumes and files
nixos-rebuild *                                                # activates a system, also behind `ssh <guest>`
zfs create destroy set snapshot rollback                       # dataset writes
zfs rename clone promote inherit receive recv                  # dataset writes
zpool create destroy add remove attach detach replace split    # pool layout; `list` and `status` read
zpool import export online offline clear labelclear            # pool state
zpool scrub trim initialize set upgrade                        # pool work and properties
pvesm add remove set alloc free                                # storage definitions and volumes; `status` reads
vzdump *                                                       # writes a backup
ha-manager add remove set migrate relocate crm-command rules   # HA resources; `status` and `config` read
systemctl start stop restart try-restart reload                # unit state; `is-active`, `show`, `status` read
systemctl reload-or-restart enable disable mask unmask         # unit state and boot links
systemctl daemon-reload reboot poweroff halt kill              # the manager and the machine
reboot *                                                       # the machine
poweroff *                                                     # the machine
qm guest                                                       # runs a command inside a guest, read or write cannot be told apart
pct exec                                                       # the same in a container
pvecm add delnode expected                                     # cluster membership; `status` and `nodes` read
dd *                                                           # writes a device
wipefs *                                                       # erases a disk's signatures
sgdisk *                                                       # rewrites a partition table
apt-get install remove purge upgrade dist-upgrade              # packages on a node
scp *                                                          # copies files onto the other machine, or fetches over a local file
rsync *                                                        # the same
ssh-copy-id *                                                  # installs a key on the other machine
ssh-keygen -R                                                  # edits known_hosts
curl -X --request -d --data --data-raw --data-binary --json    # a body or a method: an API write (OPNsense, UniFi, a module's); a plain GET reads
curl -F --form -T --upload-file                                # uploads
git commit push pull fetch merge rebase reset checkout switch  # moves refs or the working tree; `log`, `show`, `status` read
git clean clone init restore revert cherry-pick am apply       # the same
git add rm mv gc submodule                                     # the same
git stash:push stash:pop stash:apply stash:drop stash:clear    # `stash list` and `stash show` read
git worktree:add worktree:remove worktree:prune                # `worktree list` reads
git remote:add remote:remove remote:set-url                    # `remote -v` reads
nix copy                                                       # puts a closure on another machine
nix profile                                                    # changes a profile
nix store:delete store:gc                                      # deletes from the store
nix --out-link -o                                              # a build that links a gcroot or a component; a plain `nix build` only fills the store
nix-env *                                                      # changes a profile, the system's too
nix-collect-garbage *                                          # deletes from the store
switch-to-configuration *                                      # activates a system; update-os.sh runs it directly
node-provisioner register unregister enable disable           # arms a node's install; `assets`, `list`, `status` read
systemd-cat *                                                  # writes the journal, which the event stream reads
sudo rm mv cp tee install ln chmod chown mkdir touch truncate sed  # file tools as root: outside config/ too; `sudo cat` reads
